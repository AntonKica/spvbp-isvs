\chapter{Zavádzanie ISMS} \label{ISMS}
Čo to znamená zaviesť ISMS a udržiavať ho?

\section*{Prevzatie celkovej zodpovednosti vedením organizácie za KIB}

\begin{enumerate}
\item Vedenie organizácie MUSÍ prevziať celkovú zodpovednosť za KIB, tak aby to bolo jasné každému.
\item Vedenie organizácie MUSÍ spustiť, riadiť a monitorovať bezpečnostný proces.
\item Vedenie organizácie MUSÍ byť dobrým príkladom pri dodržiavaní zásad KIB.
\item Vedenie organizácie MUSÍ definovať zodpovednosť za KIB.
\item Zodpovední zamestnanci MUSIA mať potrebnú kvalifikáciu a zdroje.
\item Vedenie organizácie MUSÍ byť v pravidelných intervaloch informované o stave KIB v organizácii.
\item Vedenie organizácie MUSÍ byť informované o možných rizikách a dôsledkoch nedostatočných bezpečnostných opatrení.
\end{enumerate}

\section*{Určenie cieľov a stratégie KIB}
\begin{enumerate}
\item Vedenie organizácie MUSÍ spustiť a ustanoviť bezpečnostný proces.
\item Vedenie organizácie MUSÍ definovať a dokumentovať vhodné ciele a stratégiu KIB.
\item Koncepčné materiály KIB MUSIA byť vypracované.
\item Vedenie organizácie MUSÍ prevziať zodpovednosť a podporovať ciele a stratégiu KIB.
\item Vedenie organizácie MUSÍ v pravidelných intervaloch revidovať ciele a stratégiu KIB.
\end{enumerate}

\section*{Načrtnutie politiky KIB}
\begin{enumerate}
\item Vedenie organizácie MUSÍ prijať zastrešujúcu politiku KIB.
\item Politika KIB MUSÍ popísať hodnotu KIB, ciele KIB, dôležité prvky stratégie KIB a organizačnú štruktúru KIB.
\item Rozsah politiky KIB MUSÍ byť jasne určený.
\item Politiky KIB MUSÍ vysvetliť ciele KIB, a ako súvisia s poslaním organizácie.
\item Vedenie organizácie MUSÍ oboznámiť všetkých zamestnancov o politike KIB.
\item Politiky KIB by MALA byť revidovaná v pravidelných intervaloch.
\end{enumerate}

\section*{Vymenovanie manažéra KIB}
\begin{enumerate}
\item Vedenie organizácie MUSÍ vymenovať manažéra KIB.
\item Manažér KIB MUSÍ podporovať KIB a pomáhať riadiť bezpečnostný proces.
\item Vedenie organizácie MUSÍ poskytnúť manažérovi KIB dostatočné zdroje.
\item Manažér KIB MUSÍ mať možnosť komunikovať priamo s vedením organizácie.
\item Manažér KIB MUSÍ byť prítomnú v rannom štádiu zavádzaní nových aplikácií a IT systémov.
\end{enumerate}

\section*{Vytvorenie vhodnej organizačnej štruktúry KIB}
\begin{enumerate}
\item Organizácia MUSÍ mať vhodnú vysokoúrovňovú organizačnú štruktúru KIB.
\item MUSIA byť vytvorené nové role KIB pre špecifické úlohy KIB.
\item Do rolí MUSIA byť zaradení kvalifikovaní zamestnanci s dostatočnými zdrojmi.
\item Roly, úlohy, zodpovednosti a kompetencia MUSIA byť transparentne priradené.
\item MUSÍ byť stanovené efektívne pravidlá zastupovania pre všetky dôležité časti organizácie.
\item Komunikačné kanály MUSIA byť naplánované, opísané, zavedené a zverejnené.
\item Pre všetky úlohy a role MUSÍ byť určené kto má koho informovať, o akých úkonoch a čo všetko má vedieť.
\item Organizačná štruktúra KIB musí byť revidovaná v pravidelných intervaloch.
\end{enumerate}

\section*{Definícia bezpečnostných opatrení}
\begin{enumerate}
\item Pre všetky aspekty spracovania informácie MUSIA byť definované detailné a primerané bezpečnostné opatrenia.
\item Všetky bezpečnostné opatrenia by MALI byť systematický dokumentované v bezpečnostných konceptoch.
\item Bezpečnostné opatrenia by MALI byť pravidelne revidované.
\end{enumerate}

\section*{Zapojenie zamestnancov do bezpečnostného procesu}
\begin{enumerate}
\item Všetci zamestnanci MUSIA byť integrovaní do bezpečnostného procesu.
\item Zamestnanci MUSIA byť informovaní o nebezpečenstvách, ktoré sú pre nich relevantné.
\item Zamestnanci MUSIA vedieť, ako a môcť implementovať bezpečnostné opatrenia súvisiace s ich pracoviskom.
\item Zamestnancom MUSÍ byť umožnené aktívne prispievať k bezpečnosti.
\item Zamestnancom by MALI byť prítomný v rannom štádiu plánovania bezpečnostných opatrení.
\item Pri zavádzaní bezpečnostných politík a nástrojov zamestnanci MUSIA vhodne informovaný, ako s nimi majú zaobchádzať.
\item Zamestnanci MUSIA byť oboznámení s následkami porušenia bezpečnostných pravidiel.
\end{enumerate}

\section*{Integrovanie KIB do procedúr a procesov celej organizácie}
\begin{enumerate}
\item KIB MUSÍ byť integrovaná do všetkých biznisových procesov a špecializovaných úloh.
\item Všetky bezpečnostné aspekty MUSIA byť zohľadnené pri prebiehajúcich činnostiach a aj pri nových činnostiach organizácie.
\item Manažér KIB sa musí podieľať pri všetkých rozhodnutiach súvisiacich s bezpečnosťou.
\item KIB by mala byť koordinovaná s ďalšími oblasťami organizácie zaoberajúce sa riadením rizík a bezpečnosti.
\end{enumerate}
