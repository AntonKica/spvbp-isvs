\section{KIB}
%% [00]
\begin{frame}
  \frametitle{Čo to je kybernetická a informačná bezpečnosť?}
  Kybernetická a informačná bezpečnosť je multidisciplinárna oblasť ľudskej činnosti\cite{KUdIKBaMVS}, ktorá skúma:
  \begin{itemize}
    \item hrozby voči informáciám, informačným systémom a sieťam,
    \item riešenia eliminácie riziká a hrozieb,
    \item metódy dlhodobej a efektívnej ochrany aktív organizácie.
  \end{itemize}
\end{frame}

%% [00]
\begin{frame}
  \frametitle{Prečo by nás mala zaujímať KIB?}

  Dôvodov je mnoho, najmä zaručiť:

  \begin{itemize}
    \item dôvernosť, integritu a dostupnosť informácie a informačných systémov,
    \item spoľahlivosť systémov a spracovanej informácie,
    \item bezpečnosť okolitých systémov.
  \end{itemize}
\end{frame}

    \item 95/2019 Z. z., zákon o informačných technológiách vo verejnej správe
  \end{itemize}
\end{frame}

\section{Právne normy}
\begin{frame}
  \frametitle{Zákon o kybernetickej bezpečnosti}
  %Všetko to začalo informatizáciou spoločnosti\dots \cite{NBU-VS}
  \begin{itemize}
    \item kompetenčný zákon, \textit{lex generalis},
    \item prevádzkovateľ základnej služby,
    \item rámcovo určuje 16 oblastí, pre ktoré má prevádzkovateľ prijať bezpečnostné opatrenia,
    \item určenie manažéra kybernetickej a informačnej bezpečnosti,
    \item upravuje povinnosti tretích strán,
    \item vyhláška 362/2018 Z.z. -- rozsah bezpečnostnej dokumentácie, kategorizácia sieti a informačných systémov.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Zákon o informačných technológiách vo verejnej správe}
  %Všetko to začalo informatizáciou spoločnosti\dots \cite{NBU-VS}
  \begin{itemize}
    \item \textit{lex speciallis}, sektor verejná správa, podsektor informačné systémy,
    \item správca informačného systému verejnej správy,
    \item zavedenie systému riadenia informačnej bezpečnosti,
    \item možnosť postupovať pomocou medzinárodných štandardov,
    \item vyhláška 179/2020 Z.z. -- obsah bezpečnostnej dokumentácie a kategorizácia.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{To najdôležitejšie z právnych predpisov}
  \begin{itemize}
    \item určenie manažéra kybernetickej a informačnej bezpečnosti,
    \item zavedenie systému riadenia informačnej bezpečnosti.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Pomocný systém}

Pomocný systém manažérovi KIB:

  \begin{itemize}
    \item objasní, prečo sa musia v organizácii zaoberať KIB,
    \item poskytne minimálne znalosti KIB,
    \item vysvetlí a prevedie ho procesom zavádzania ISMS,
